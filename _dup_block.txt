// FX Rates (P07 Finance) — Production schema
// ============================================
app.get("/api/v1/fx-rates", requireAuth, async (req: express.Request, res: express.Response) => {
  try {
    const { rows } = await req.db.query(
      `SELECT id, from_currency_id, to_currency_id, rate, rate_type, rate_source, rate_date, is_active, created_at, updated_at
       FROM fx_rates WHERE is_active = true ORDER BY rate_date DESC, created_at DESC LIMIT 200`
    );
    res.json({ fxRates: rows.map((r: any) => ({
      id: r.id,
      fromCurrencyId: r.from_currency_id,
      toCurrencyId: r.to_currency_id,
      rate: Number(r.rate),
      rateType: r.rate_type,
      rateSource: r.rate_source,
      rateDate: r.rate_date,
      isActive: r.is_active,
      createdAt: r.created_at,
      updatedAt: r.updated_at,
    })) });
  } catch (error) {
    res.status(500).json({ error: "Failed to load FX rates" });
  }
});

app.post("/api/v1/fx-rates", requireAuth, async (req: express.Request, res: express.Response) => {
  const { fromCurrencyId, toCurrencyId, rate, rateType, rateSource, rateDate } = req.body ?? {};
  if (!fromCurrencyId || !toCurrencyId || rate === undefined || !rateDate) {
    return res.status(400).json({ error: "Required fields: fromCurrencyId, toCurrencyId, rate, rateDate" });
  }
  try {
    const { rows } = await req.db.query(
      `INSERT INTO fx_rates (from_currency_id, to_currency_id, rate, rate_type, rate_source, rate_date, is_active, created_at, updated_at)
       VALUES ($1::uuid, $2::uuid, $3, $4, $5, $6::timestamptz, true, now(), now()) RETURNING *`,
      [fromCurrencyId, toCurrencyId, rate, rateType || "reference", rateSource || null, rateDate],
    );
    res.status(201).json({ fxRate: rows[0] });
  } catch (error) {
    res.status(500).json({ error: "Failed to create FX rate" });
  }
});

app.post("/api/v1/journal-entries/:id/post", requireAuth, async (req: express.Request, res: express.Response) => {
  const journalEntryId = req.params.id;
  try {
    const { rows: entryRows } = await req.db.query("SELECT id, status FROM journal_entries WHERE id = $1::uuid", [journalEntryId]);
    if (entryRows.length === 0) return res.status(404).json({ error: "Journal entry not found" });
    if (entryRows[0].status !== 'draft') return res.status(400).json({ error: "Only draft journal entries can be posted" });
    const { rows: lines } = await req.db.query(
      `SELECT id, account_id, debit, credit FROM journal_lines WHERE journal_entry_id = $1::uuid ORDER BY line_order`,
      [journalEntryId],
    );
    if (lines.length === 0) return res.status(400).json({ error: "Cannot post a journal entry with no lines" });
    const totalDebit = lines.reduce((sum: number, l: any) => sum + Number(l.debit || 0), 0);
    const totalCredit = lines.reduce((sum: number, l: any) => sum + Number(l.credit || 0), 0);
    if (Math.abs(totalDebit - totalCredit) > 0.005) return res.status(400).json({ error: "Journal entry is not balanced", totalDebit, totalCredit });
    const { rows } = await req.db.query(
      `UPDATE journal_entries SET status = 'posted', posted_at = now(), posted_by = $2::uuid, updated_at = now()
       WHERE id = $1::uuid AND status = 'draft' RETURNING *`,
      [journalEntryId, req.user?.id || null],
    );
    if (rows.length === 0) return res.status(409).json({ error: "Journal entry was modified by another operation" });
    res.json({ journalEntry: rows[0] });
  } catch (error) {
    res.status(500).json({ error: "Failed to post journal entry" });
  }
});

app.post("/api/v1/journal-lines", requireAuth, async (req: express.Request, res: express.Response) => {
  const { journalEntryId, accountId, debit, credit, description, lineOrder } = req.body ?? {};
  if (!journalEntryId || !accountId || debit === undefined || credit === undefined) {
    return res.status(400).json({ error: "journalEntryId, accountId, debit, credit required" });
  }
  try {
    const { rows: entryRows } = await req.db.query("SELECT id, status FROM journal_entries WHERE id = $1::uuid", [journalEntryId]);
    if (entryRows.length === 0) return res.status(404).json({ error: "Journal entry not found" });
    if (entryRows[0].status !== 'draft') return res.status(400).json({ error: "Cannot add lines to a posted journal entry" });
    const { rows: accountRows } = await req.db.query("SELECT id, is_active FROM chart_of_accounts WHERE id = $1::uuid", [accountId]);
    if (accountRows.length === 0) return res.status(400).json({ error: "Invalid account: account not found in chart of accounts" });
    if (!accountRows[0].is_active) return res.status(400).json({ error: "Invalid account: account is not active" });
    const { rows } = await req.db.query(
      `INSERT INTO journal_lines (journal_entry_id, account_id, debit, credit, description, line_order, created_at)
       VALUES ($1::uuid, $2::uuid, $3, $4, $5, $6, now()) RETURNING *`,
      [journalEntryId, accountId, debit || 0, credit || 0, description || null, lineOrder !== undefined ? lineOrder : 1],
    );
    res.status(201).json({ journalLine: rows[0] });
  } catch (error) {
    res.status(500).json({ error: "Failed to create journal line" });
  }
});
